local function runGameScript()
    if isRunning then return end
    isRunning = true

    local currentPlaceId = game.PlaceId
    local scriptUrl = SupportedGames[currentPlaceId]
    
    if scriptUrl and scriptUrl ~= "" then
        if scriptUrl == "SPECIAL:StealAnEgg" then
            showStealAnEggVersionSelector()
        else
            local success, scriptContent = pcall(function()
                return game:HttpGet(scriptUrl)
            end)

            if success and scriptContent then
                local loadSuccess, loadError = pcall(function()
                    loadstring(scriptContent)()
                end)
                if not loadSuccess then
                    warn("[" .. HubName .. "] Error running script: " .. tostring(loadError))
                end
            else
                warn("[" .. HubName .. "] Failed to fetch script from GitHub.")
            end
        end
    else
        warn("[" .. HubName .. "] Game not supported. PlaceId: " .. tostring(currentPlaceId))
        showDiscordGUI()
    end
    
    isRunning = false
end

-- Initial run
runGameScript()

-- Watch for teleport
LocalPlayer.OnTeleport:Connect(function(state)
    if state == Enum.TeleportState.Started then
        isRunning = false
        task.wait(2)
        runGameScript()
    elseif state == Enum.TeleportState.Failed then
        isRunning = false
    end
end)

-- Fallback check every 5 seconds
task.spawn(function()
    while true do
        task.wait(5)
        if game.PlaceId ~= lastPlaceId then
            lastPlaceId = game.PlaceId
            isRunning = false
            task.wait(1)
            runGameScript()
        end
    end
end)
